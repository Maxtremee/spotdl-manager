    ```typescript
    import { randomUUID } from "node:crypto";
    import { eq, sql } from "drizzle-orm";
    import type { AppLogger } from "~/logger";
    import { Logger } from "~/logger";
    import { getDb, schema, type TDatabase } from "../db";

    export interface NormalizedTrack {
      spotifyTrackId: string;
      title: string;
      artist: string;
      durationMs: number;
      position: number;
    }

    /**
     * Scrape-side DB writes. Both methods are callable standalone, but
     * `upsertAll` wraps both in a single sync transaction so a failure in either
     * leg rolls the whole scrape back (D-08 atomicity).
     */
    export class ScraperRepository {
      private readonly db: TDatabase;
      private readonly logger: AppLogger;

      constructor(db: TDatabase = getDb(), logger: AppLogger = Logger.get("ScraperRepository")) {
        this.db = db;
        this.logger = logger;
      }

      /**
       * Atomic: tracks upsert + sources.coverArtUrl update in one transaction.
       * On conflict (source_id, spotify_track_id): refresh title/artist/durationMs/position/updatedAt ONLY.
       * Never touches state, ytVideoId, downloadPath, failureReason (D-14).
       */
      async upsertAll(sourceId: string, tracks: NormalizedTrack[], coverArtUrl: string | null): Promise<void> {
        this.db.transaction((tx) => {
          if (tracks.length > 0) {
            const now = new Date();
            const values = tracks.map((t) => ({
              id: randomUUID(),
              sourceId,
              spotifyTrackId: t.spotifyTrackId,
              title: t.title,
              artist: t.artist,
              durationMs: t.durationMs,
              position: t.position,
              createdAt: now,
              updatedAt: now,
            }));
            tx
              .insert(schema.tracks)
              .values(values)
              .onConflictDoUpdate({
                target: [schema.tracks.sourceId, schema.tracks.spotifyTrackId],
                set: {
                  title: sql.raw(`excluded.${schema.tracks.title.name}`),
                  artist: sql.raw(`excluded.${schema.tracks.artist.name}`),
                  durationMs: sql.raw(`excluded.${schema.tracks.durationMs.name}`),
                  position: sql.raw(`excluded.${schema.tracks.position.name}`),
                  updatedAt: sql.raw(`excluded.${schema.tracks.updatedAt.name}`),
                  // D-14: NO state, ytVideoId, downloadPath, failureReason here — omission preserves existing values.
                },
              })
              .run();
          }
          // D-08: cover_art_url in same txn. Only update if we have a value;
          // nulls (playlist without image) don't need to overwrite an existing URL
          // — but we also must handle explicit null from the envelope (no-op is safe).
          if (coverArtUrl !== null) {
            tx
              .update(schema.sources)
              .set({ coverArtUrl, updatedAt: new Date() })
              .where(eq(schema.sources.id, sourceId))
              .run();
          }
        });
      }

      /**
       * Standalone cover-art setter (used by SyncRunner in the narrow case where
       * tracks-empty but cover art still relevant; or as a unit-testing seam).
       * Not wrapped in a transaction — atomicity only matters when paired with tracks.
       */
      async setCoverArtUrl(sourceId: string, coverArtUrl: string | null): Promise<void> {
        await this.db
          .update(schema.sources)
          .set({ coverArtUrl, updatedAt: new Date() })
          .where(eq(schema.sources.id, sourceId));
      }
    }
    ```

    **IMPORTANT — Drizzle better-sqlite3 transaction shape:** better-sqlite3 is synchronous by nature. Drizzle's `.transaction` on sqlite accepts a synchronous callback. Check the current code for prior transaction usage (there may be none in Phase 1); if the API mismatches, consult Context7 `/drizzle-team/drizzle-orm-docs` for better-sqlite3 transactions. The sketch above uses sync callback + `.run()` terminators on each builder — the correct idiom for better-sqlite3.

    If `db.transaction` turns out to require a non-async body AND the callback's `.run()` calls cannot be awaited, the outer method signature can stay `async` (the Promise resolves when the sync transaction completes — Drizzle returns a value from `db.transaction` synchronously with better-sqlite3, and the method can return it wrapped in `Promise.resolve`). Verify during implementation.

    ```typescript
    import { randomUUID } from "node:crypto";
    import type { AppLogger } from "~/logger";
    import { Logger } from "~/logger";
    import type { SourceRow } from "../db/schema";
    import { getEventBus } from "../events";
    import type { FailureReason } from "../events/schema";
    import { InvocationRepository } from "../invocation/repository";
    import type { PythonEnvelope } from "./schema";
    import { ScraperRepository } from "./repository";
    import { SpotifyScraperBridge } from "./SpotifyScraperBridge";

    export interface SyncRunnerDeps {
      bridge?: SpotifyScraperBridge;
      invocationRepo?: InvocationRepository;
      scraperRepo?: ScraperRepository;
      logger?: AppLogger;
      // Escape hatch for deterministic tests:
      clock?: () => Date;
    }

    const TRUNCATION_THRESHOLD = 100;  // Spotify embed cap — spike 002

    export function isValidPlaylistUrl(url: string): boolean {
      try {
        const u = new URL(url);
        return u.hostname === "open.spotify.com" && u.pathname.startsWith("/playlist/");
      } catch {
        return false;
      }
    }

    export class SyncRunner {
      private readonly bridge: SpotifyScraperBridge;
      private readonly invocationRepo: InvocationRepository;
      private readonly scraperRepo: ScraperRepository;
      private readonly logger: AppLogger;
      private readonly clock: () => Date;

      constructor(deps: SyncRunnerDeps = {}) {
        this.bridge = deps.bridge ?? new SpotifyScraperBridge();
        this.invocationRepo = deps.invocationRepo ?? new InvocationRepository();
        this.scraperRepo = deps.scraperRepo ?? new ScraperRepository();
        this.logger = deps.logger ?? Logger.get("SyncRunner");
        this.clock = deps.clock ?? (() => new Date());
      }

      async run(source: SourceRow): Promise<void> {
        const invocationId = randomUUID();
        const startedAt = this.clock();
        const eventBus = getEventBus();

        try {
          // Emit started first — metrics/webhook consumers see the lifecycle begin.
          await eventBus.emit({
            type: "playlist.sync.started",
            payload: {
              playlistId: source.id,
              playlistName: source.name,
              invocationId,
              sourceUrl: source.sourceUrl,
              outputDir: source.outputDir,
            },
          });

          await this.invocationRepo.create({
            id: invocationId,
            playlistId: source.id,
            startedAt,
            status: "running",
          });

          // T-2-04: defense-in-depth URL guard (client already validates; re-check server-side).
          if (!isValidPlaylistUrl(source.sourceUrl)) {
            await this.finalizeFailure(invocationId, source, "invalid_url",
              `URL not a Spotify playlist: ${source.sourceUrl}`);
            return;
          }

          const envelope = await this.bridge.fetchPlaylist(source.sourceUrl);

          if (envelope.error) {
            // Detect python_crash synthesis by the bridge (message prefix convention).
            const isPythonCrash =
              envelope.tracks === null &&
              envelope.error.type === "network_error" &&
              envelope.error.message.startsWith("python crash:");
            const failureReason: FailureReason = isPythonCrash
              ? "python_crash"
              : envelope.error.type;
            await this.finalizeFailure(invocationId, source, failureReason, envelope.error.message);
            return;
          }

          if (!envelope.tracks) {
            await this.finalizeFailure(invocationId, source, "python_crash",
              "python returned no tracks without error");
            return;
          }

          const truncationSuspected = envelope.tracks.length >= TRUNCATION_THRESHOLD;

          try {
            await this.scraperRepo.upsertAll(
              source.id,
              envelope.tracks.map((t) => ({
                spotifyTrackId: t.spotify_track_id,
                title: t.title,
                artist: t.artist,
                durationMs: t.duration_ms,
                position: t.position,
              })),
              envelope.cover_art_url,
            );
          } catch (err) {
            this.logger.error({ err, sourceId: source.id }, "Upsert failed");
            // D-08 honored by the repository's transaction; we just classify.
            await this.finalizeFailure(invocationId, source, "python_crash",
              err instanceof Error ? err.message : String(err));
            return;
          }

          await this.finalizeSuccess(invocationId, source, envelope.tracks.length, truncationSuspected, startedAt);
        } catch (unexpected) {
          // Pitfall 8: always emit a terminal event.
          this.logger.error({ err: unexpected, sourceId: source.id }, "SyncRunner crash");
          try {
            await this.finalizeFailure(invocationId, source, "python_crash",
              unexpected instanceof Error ? unexpected.message : String(unexpected));
          } catch (terminalErr) {
            this.logger.error({ err: terminalErr }, "Failed to emit terminal event — lifecycle broken");
          }
        }
      }

      private async finalizeSuccess(
        invocationId: string, source: SourceRow, trackCount: number,
        truncationSuspected: boolean, startedAt: Date,
      ): Promise<void> {
        const finishedAt = this.clock();
        const duration = Math.max(1, finishedAt.getTime() - startedAt.getTime());

        await this.invocationRepo.update(invocationId, {
          finishedAt,
          exitCode: 0,
          status: "success",
          summary: JSON.stringify({ track_count: trackCount, truncation_suspected: truncationSuspected }),
        });

        await getEventBus().emit({
          type: "playlist.sync.completed",
          payload: {
            playlistId: source.id,
            playlistName: source.name,
            invocationId,
            duration,
            exitCode: 0,
            trackCount,
            truncationSuspected,
          },
        });
      }

      private async finalizeFailure(
        invocationId: string, source: SourceRow,
        failureReason: FailureReason, errorMessage: string,
      ): Promise<void> {
        await this.invocationRepo.update(invocationId, {
          finishedAt: this.clock(),
          exitCode: 1,
          status: "failed",
          summary: JSON.stringify({ failure_reason: failureReason, error: errorMessage }),
        });

        await getEventBus().emit({
          type: "playlist.sync.failed",
          payload: {
            playlistId: source.id,
            playlistName: source.name,
            invocationId,
            error: errorMessage,
            failureReason,
          },
        });
      }
    }
    ```
